    62→    62→// Helper: check if a jalur name is Non-Akademik variant
    63→    63→function isNonAkademikJalur(subJalur: string): boolean {
    64→    64→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    65→    65→  return lower.includes('nonakademik')
    66→    66→}
    67→    67→
    68→    68→// Helper: check if a jalur name is Prestasi Akademik variant
    69→    69→function isAkademikJalur(subJalur: string): boolean {
    70→    70→  const lower = subJalur.toLowerCase()
    71→    71→  return lower.includes('prestasi') && !lower.includes('non')
    72→    72→}
    73→    73→
    74→    74→// Helper: get the display label for the prestasi score column
    75→    75→function getPrestasiScoreLabel(subJalur: string): string {
    76→    76→  if (isNonAkademikJalur(subJalur)) return 'Skor Pres. Non-Akd'
    77→    77→  if (isAkademikJalur(subJalur)) return 'Skor Pres. Akd'
    78→    78→  return 'Skor Prestasi'
    79→    79→}
    80→    80→
    81→    81→// Helper: get the prestasi score for display from a record
    82→    82→function getPrestasiDisplayValue(r: Record<string, unknown>): string {
    83→    83→  const subJalur = (r.subJalur as string) || ''
    84→    84→  if (isNonAkademikJalur(subJalur)) {
    85→    85→    return (r.skorPrestasiNonAkademik as string) || (r.skorPrestasiAkademik as string) || '-'
    86→    86→  }
    87→    87→  return (r.skorPrestasiAkademik as string) || (r.skorPrestasiNonAkademik as string) || '-'
    88→    88→}
    89→    89→
    90→    90→// Helper: get the numeric prestasi score for sorting
    91→    91→function getPrestasiNumValue(r: Record<string, unknown>): number {
    92→    92→  const subJalur = (r.subJalur as string) || ''
    93→    93→  const akdNum = (r._skorPrestasiAkademikNum as number) || -1
    94→    94→  const nonAkdNum = (r._skorPrestasiNonAkademikNum as number) || -1
    95→    95→
    96→    96→  if (isNonAkademikJalur(subJalur)) {
    97→    97→    return nonAkdNum > 0 ? nonAkdNum : akdNum
    98→    98→  }
    99→    99→  return akdNum > 0 ? akdNum : nonAkdNum
   100→   100→}
