     4→     4→/**

     6→     6→ */
     7→     7→function v(r: Record<string, unknown>, key: string): string {
     8→     8→  const val = r[key]
     9→     9→  if (val === null || val === undefined) return '-'
    10→    10→  return String(val) || '-'
    11→    11→}

    13→    13→// Helper: check if a jalur name is Non-Akademik variant
    14→    14→function isNonAkademikJalur(subJalur: string): boolean {
    15→    15→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    16→    16→  return lower.includes('nonakademik')
    17→    17→}

    19→    19→// Helper: get prestasi score for display from a record
    20→    20→function getPrestasiDisplayValue(r: Record<string, unknown>): string {
    21→    21→  const subJalur = (r.subJalur as string) || ''
    22→    22→  if (isNonAkademikJalur(subJalur)) {
    23→    23→    return v(r, 'skorPrestasiNonAkademik') !== '-' ? v(r, 'skorPrestasiNonAkademik') : v(r, 'skorPrestasiAkademik')
    24→    24→  }
    25→    25→  return v(r, 'skorPrestasiAkademik') !== '-' ? v(r, 'skorPrestasiAkademik') : v(r, 'skorPrestasiNonAkademik')
    26→    26→}

    28→    28→// Helper: get numeric prestasi score from a record
    29→    29→function getPrestasiNumValue(r: Record<string, unknown>): number {
    30→    30→  const subJalur = (r.subJalur as string) || ''
    31→    31→  const akdNum = (r._skorPrestasiAkademikNum as number) || -1
    32→    32→  const nonAkdNum = (r._skorPrestasiNonAkademikNum as number) || -1
    33→    33→  if (isNonAkademikJalur(subJalur)) {
    34→    34→    return nonAkdNum > 0 ? nonAkdNum : akdNum
    35→    35→  }
    36→    36→  return akdNum > 0 ? akdNum : nonAkdNum
    37→    37→}

   358→   358→export function handleRankingExportExcel(params: {
   359→   359→  rankingPreviewJalur: string
   360→   360→  rankingTampilan: string
   361→   361→  rankingSekolah: string
   362→   362→  rankingJurusan: string
   363→   363→  rankingStatus: string
   364→   364→  rankingData: Array<Record<string, unknown>>
   365→   365→  rankingKuota: number
   366→   366→  rankingKuotaPerJalur: Array<{ nama: string; persentase: number; kuota: number }>
   367→   367→  appName: string
   368→   368→  schoolName: string
   369→   369→}): void {
   370→   370→  const {
   371→   371→    rankingPreviewJalur: selectedJalur,
   372→   372→    rankingTampilan,
   373→   373→    rankingSekolah,
   374→   374→    rankingJurusan,
   375→   375→    rankingStatus,
   376→   376→    rankingData,
   377→   377→    rankingKuota,
   378→   378→    rankingKuotaPerJalur,
   379→   379→    appName,
   380→   380→    schoolName,
   381→   381→  } = params
   382→   382→
   383→   383→  const sortLabel = rankingTampilan === 'jarak' ? 'Jarak Terdekat' : rankingTampilan === 'nilai' ? 'Nilai Tertinggi' : rankingTampilan === 'komposit' ? 'Skor Komposit' : 'Skor Prestasi'
   384→   384→  const jalurLabel = selectedJalur !== 'all' ? selectedJalur : 'Semua Jalur'
   385→   385→
   386→   386→  // Filter data by selected jalur
   387→   387→  const filteredData = selectedJalur === 'all'
   388→   388→    ? rankingData
   389→   389→    : rankingData.filter((r: Record<string, unknown>) => (r.subJalur as string) === selectedJalur)
   390→   390→

   394→   394→    const jalur = r.subJalur as string
   395→   395→    if (!jalurRankCounters[jalur]) jalurRankCounters[jalur] = 0
   396→   396→    jalurRankCounters[jalur]++
   397→   397→    return { ...r, _newRank: idx + 1, _newJalurRank: jalurRankCounters[jalur] }
   398→   398→  })
   399→   399→

   401→   401→    const rankNum = r._newRank
   402→   402→    const jalurRank = r._newJalurRank
   403→   403→
   404→   404→    const currentKuota = rankingKuotaPerJalur.find(k => {
   405→   405→      const jalurName = (r.subJalur as string || '').toLowerCase()
   406→   406→      return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
   407→   407→        || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
   408→   408→        || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   409→   409→        || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
   410→   410→        || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
   411→   411→    })?.kuota || 0
   412→   412→
   413→   413→    const sameJalurAbove = reRanked
   414→   414→      .filter((other: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) =>

   416→   416→        other._newJalurRank < jalurRank

   418→   418→

   420→   420→
   421→   421→    return {
   422→   422→      'Rangking': rankNum,
   423→   423→      'Rangking Jalur': jalurRank > 0 ? jalurRank : '-',

   425→   425→      'No. Registrasi': v(r, 'noRegistrasi'),

   429→   429→      'Tanggal Lahir': v(r, 'tanggalLahir'),
   430→   430→      'Sekolah Asal': v(r, 'namaSekolahAsal'),
   431→   431→      'Jurusan': v(r, 'jurusan'),
   432→   432→      'Jarak': v(r, 'lokasiJarak'),
   433→   433→      'Skor Jarak': v(r, 'skorJarak'),
   434→   434→      'Nilai Rata-Rata': v(r, 'nilaiRataRata'),
   435→   435→      'Skor Nilai Raport': v(r, 'skorNilaiRaport'),
   436→   436→      'Skor Lomba': v(r, 'skorLomba'),
   437→   437→      'Nilai Rata-Rata TKA': v(r, 'nilaiRataRataTKA'),
   438→   438→      'Skor Prestasi': v(r, 'skorPrestasiAkademik'),
   439→   439→      'Skor Prestasi Akademik': (r.skorPrestasiAkademik as string) || '-',
   440→   440→      'Skor Prestasi Non Akademik': (r.skorPrestasiNonAkademik as string) || '-',
   441→   441→      'Skor Komposit': v(r, 'skor'),
   442→   442→      'Kekurangan Verifikasi': v(r, 'kekuranganVerifikasi'),
   443→   443→      'Tanggal Verif': v(r, 'tanggalVerif'),
   444→   444→      'Jam Verif': v(r, 'jamVerif'),

   446→   446→      'Lama KK': v(r, 'lamaKK'),
   447→   447→      'Status Verifikasi': r.verificationStatus === 'VERIFIED' ? 'Diterima' : r.verificationStatus === 'REJECTED' ? 'Ditolak' : 'Menunggu',
   448→   448→      'Masuk Kuota': withinKuota ? 'Ya' : 'Tidak',
   449→   449→    }
   450→   450→  })
   451→   451→
   452→   452→  const ws = XLSX.utils.json_to_sheet(excelData)
   453→   453→

   465→   465→

   468→   468→  for (const r of reRanked) {
   469→   469→    const jalur = r.subJalur as string
   470→   470→    if (!jalurGroups[jalur]) {
   471→   471→      const kuota = rankingKuotaPerJalur.find(k => {
   472→   472→        const jalurName = jalur.toLowerCase()
   473→   473→        return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
   474→   474→          || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
   475→   475→          || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   476→   476→          || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
   477→   477→          || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
   478→   478→      })?.kuota || 0
   479→   479→      jalurGroups[jalur] = { total: 0, verified: 0, rejected: 0, pending: 0, kuota }
   480→   480→    }
   481→   481→    jalurGroups[jalur].total++
   482→   482→    if (r.verificationStatus === 'VERIFIED') jalurGroups[jalur].verified++
