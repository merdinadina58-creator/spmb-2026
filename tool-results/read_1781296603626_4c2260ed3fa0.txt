    51→// Sidebar nav item component for cleaner code
    52→function SidebarNavItem({
    53→  tab,
    54→  activeTab,
    55→  setActiveTab,
    56→  icon: Icon,
    57→  label,
    58→  badge,
    59→  colorScheme = 'emerald',
    60→  collapsed,
    61→}: {
    62→  tab: string
    63→  activeTab: string
    64→  setActiveTab: (tab: string) => void
    65→  icon: React.ComponentType<{ className?: string }>
    66→  label: string
    67→  badge?: React.ReactNode
    68→  colorScheme?: 'emerald' | 'amber' | 'red' | 'sky' | 'gray'
    69→  collapsed: boolean
    70→}) {
    71→  const colorClasses = {
    72→    emerald: {
    73→      active: 'bg-emerald-50 text-emerald-700 shadow-sm shadow-emerald-100/50',
    74→      activeIcon: 'text-emerald-600',
    75→      activeBar: 'before:bg-emerald-500',
    76→    },
    77→    amber: {
    78→      active: 'bg-amber-50 text-amber-700 shadow-sm shadow-amber-100/50',
    79→      activeIcon: 'text-amber-600',
    80→      activeBar: 'before:bg-amber-500',
    81→    },
    82→    red: {
    83→      active: 'bg-red-50 text-red-700 shadow-sm shadow-red-100/50',
    84→      activeIcon: 'text-red-600',
    85→      activeBar: 'before:bg-red-500',
    86→    },
    87→    sky: {
    88→      active: 'bg-sky-50 text-sky-700 shadow-sm shadow-sky-100/50',
    89→      activeIcon: 'text-sky-600',
    90→      activeBar: 'before:bg-sky-500',
    91→    },
    92→    gray: {
    93→      active: 'bg-gray-100 text-gray-700 shadow-sm',
    94→      activeIcon: 'text-gray-600',
    95→      activeBar: 'before:bg-gray-500',
    96→    },
    97→  }
    98→
    99→  const isActive = activeTab === tab
   100→  const colors = colorClasses[colorScheme]
   101→
   102→  const button = (
   103→    <button
   104→      onClick={() => setActiveTab(tab)}
   105→      className={`w-full flex items-center gap-3 rounded-lg text-sm transition-all duration-200 relative group
   106→        ${collapsed ? 'px-2 py-2.5 justify-center' : 'px-3 py-2.5'}
   107→        ${isActive
   108→          ? `${colors.active} font-medium before:absolute before:left-0 before:top-1/2 before:-translate-y-1/2 before:w-[3px] before:h-5 before:rounded-full ${colors.activeBar} ${collapsed ? 'before:-ml-2' : 'before:-ml-3'}`
   109→          : 'text-gray-600 hover:bg-gray-50 hover:text-gray-900'
   110→        }`}
   111→    >
   112→      <Icon className={`w-4 h-4 shrink-0 ${isActive ? colors.activeIcon : 'text-gray-400 group-hover:text-gray-600'}`} />
   113→      {!collapsed && (
   114→        <>
   115→          <span className="flex-1 text-left">{label}</span>
   116→          {badge}
   117→        </>
   118→      )}
   119→    </button>
   120→  )
   121→
   122→  if (collapsed) {
   123→    return (
   124→      <Tooltip delayDuration={0}>
   125→        <TooltipTrigger asChild>
   126→          {button}
   127→        </TooltipTrigger>
   128→        <TooltipContent side="right" sideOffset={8} className="flex items-center gap-2">
   129→          {label}
   130→          {badge}
   131→        </TooltipContent>
   132→      </Tooltip>
   133→    )
   134→  }
   135→
   136→  return button
   137→}
