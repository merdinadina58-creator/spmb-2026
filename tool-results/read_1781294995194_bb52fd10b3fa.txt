     4→/**
     5→ * Safely get a string value from ranking data record
     6→ */
     7→function v(r: Record<string, unknown>, key: string): string {
     8→  const val = r[key]
     9→  if (val === null || val === undefined) return '-'
    10→  return String(val) || '-'
    11→}
    12→
    13→/**
    14→ * Generate print HTML content for ranking report.
    15→ * Designed to match the detail level of Lembar Verifikasi.
    16→ */
    17→export function getRankingPrintHTML(params: {
    18→  selectedJalur: string
    19→  rankingTampilan: string
    20→  rankingSekolah: string
    21→  rankingJurusan: string
    22→  rankingStatus: string
    23→  rankingData: Array<Record<string, unknown>>
    24→  rankingKuota: number
    25→  rankingKuotaPerJalur: Array<{ nama: string; persentase: number; kuota: number }>
    26→  appName: string
    27→  schoolName: string
    28→}): string {
    29→  const {
    30→    selectedJalur,
    31→    rankingTampilan,
    32→    rankingSekolah,
    33→    rankingJurusan,
    34→    rankingStatus,
    35→    rankingData,
    36→    rankingKuota,
    37→    rankingKuotaPerJalur,
    38→    appName,
    39→    schoolName,
    40→  } = params
    41→
    42→  const sortLabel = rankingTampilan === 'jarak' ? 'Jarak Terdekat' : rankingTampilan === 'nilai' ? 'Nilai Tertinggi' : 'Skor Komposit'
    43→  const jalurLabel = selectedJalur !== 'all' ? selectedJalur : 'Semua Jalur'
    44→  const sekolahLabel = rankingSekolah !== 'all' ? rankingSekolah : 'Semua Sekolah'
    45→  const jurusanLabel = rankingJurusan !== 'all' ? rankingJurusan : 'Semua Jurusan'
    46→  const statusLabel = rankingStatus !== 'all' ? (rankingStatus === 'VERIFIED' ? 'Diterima' : rankingStatus === 'REJECTED' ? 'Ditolak' : 'Menunggu') : 'Semua Status'
    47→
    48→  // Filter data by selected jalur
    49→  const filteredData = selectedJalur === 'all'
    50→    ? rankingData
    51→    : rankingData.filter((r: Record<string, unknown>) => (r.subJalur as string) === selectedJalur)
    52→
    53→  // Re-rank within selected jalur
    54→  const jalurRankCounters: Record<string, number> = {}
    55→  const reRanked = filteredData.map((r: Record<string, unknown>, idx: number) => {
    56→    const jalur = r.subJalur as string
    57→    if (!jalurRankCounters[jalur]) jalurRankCounters[jalur] = 0
    58→    jalurRankCounters[jalur]++
    59→    return { ...r, _newRank: idx + 1, _newJalurRank: jalurRankCounters[jalur] }
    60→  })
    61→
    62→  // Helper to find kuota for a jalur
    63→  function findKuota(subJalur: string): number {
    64→    return rankingKuotaPerJalur.find(k => {
    65→      const jalurName = subJalur.toLowerCase()
    66→      return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
    67→        || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
    68→        || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
    69→        || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
    70→        || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
    71→    })?.kuota || 0
    72→  }
    73→
    74→  // Build per-jalur summary stats
    75→  const jalurGroups: Record<string, { total: number; verified: number; rejected: number; pending: number; kuota: number }> = {}

    77→    const jalur = r.subJalur as string

    79→      jalurGroups[jalur] = { total: 0, verified: 0, rejected: 0, pending: 0, kuota: findKuota(jalur) }
    80→    }
    81→    jalurGroups[jalur].total++
    82→    if (r.verificationStatus === 'VERIFIED') jalurGroups[jalur].verified++
    83→    else if (r.verificationStatus === 'REJECTED') jalurGroups[jalur].rejected++
    84→    else jalurGroups[jalur].pending++
    85→  }
    86→
    87→  // Generate rows
    88→  const rows = reRanked.map((r: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) => {
    89→    const rankNum = r._newRank
    90→    const jalurRank = r._newJalurRank
    91→    const jarakNum = r._jarakNum as number
    92→    const nilaiNum = r._nilaiNum as number
    93→    const skorNum = r._skorNum as number
    94→
    95→    // Determine kuota cutoff
    96→    const currentKuota = findKuota(r.subJalur as string)
    97→
    98→    const sameJalurAbove = reRanked
    99→      .filter((other: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) =>
   100→        (other.subJalur as string) === (r.subJalur as string) &&
   101→        other._newJalurRank < jalurRank
   102→      ).length
   103→
   104→    const withinKuota = currentKuota > 0 && sameJalurAbove < currentKuota
   105→    const isVerified = r.verificationStatus === 'VERIFIED'
   106→
   107→    const rowBg = withinKuota && !isVerified ? '#f0fdf4' : isVerified ? '#f0fdf4' : r.verificationStatus === 'REJECTED' ? '#fef2f2' : ''
   108→    const rankBg = rankNum === 1 ? '#fbbf24' : rankNum === 2 ? '#d1d5db' : rankNum === 3 ? '#b45309' : withinKuota ? '#d1fae5' : '#f3f4f6'
   109→    const rankColor = rankNum <= 3 ? '#fff' : withinKuota ? '#047857' : '#6b7280'
   110→
   111→    // Status badge
   112→    const statusBadge = r.verificationStatus === 'VERIFIED'
   113→      ? '<span style="background:#d1fae5;color:#047857;padding:1px 6px;border-radius:8px;font-size:9px">Diterima</span>'
   114→      : r.verificationStatus === 'REJECTED'
   115→        ? '<span style="background:#fee2e2;color:#b91c1c;padding:1px 6px;border-radius:8px;font-size:9px">Ditolak</span>'
   116→        : '<span style="background:#fef3c7;color:#b45309;padding:1px 6px;border-radius:8px;font-size:9px">Menunggu</span>'
   117→
   118→    // Kekurangan verifikasi
   119→    const kekurangan = v(r, 'kekuranganVerifikasi')
   120→    const kekuranganText = kekurangan !== '-'
   121→      ? kekurangan.split(' | ').map(k => `<span style="display:block;font-size:8px;color:#b91c1c;line-height:1.2">${k}</span>`).join('')
   122→      : '<span style="color:#ccc">-</span>'
   123→
   124→    // Detail Portal values
   125→    const skorLomba = v(r, 'skorLomba')
   126→    const nilaiRataRataTKA = v(r, 'nilaiRataRataTKA')
   127→    const skorPrestasiAkademik = v(r, 'skorPrestasiAkademik')
   128→
   129→    return `<tr style="background:${rowBg}">
   130→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center"><span style="display:inline-block;width:22px;height:22px;line-height:22px;border-radius:50%;background:${rankBg};color:${rankColor};font-size:9px;font-weight:bold">${rankNum}</span></td>
   131→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;line-height:1.2">${r.subJalur as string}<br><span style="font-size:8px;color:#0369a1">#${jalurRank}</span></td>
   132→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:9px;line-height:1.3"><strong>${v(r, 'nama')}</strong><br><span style="font-size:8px;color:#666">NISN: ${v(r, 'nisn')}</span></td>
   133→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;text-align:center">${v(r, 'noRegistrasi')}</td>
   134→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:9px;line-height:1.2">${v(r, 'namaSekolahAsal')}</td>
   135→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;text-align:center">${v(r, 'jurusan')}</td>
   136→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:right;font-size:9px;font-weight:${rankingTampilan === 'jarak' ? 'bold' : 'normal'};color:${jarakNum > 0 ? '#0369a1' : '#ccc'}">${v(r, 'lokasiJarak')}</td>
   137→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;color:${jarakNum > 0 ? '#0369a1' : '#ccc'}">${v(r, 'skorJarak')}</td>
   138→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:right;font-size:9px;font-weight:${rankingTampilan === 'nilai' ? 'bold' : 'normal'};color:${nilaiNum > 0 ? '#047857' : '#ccc'}">${v(r, 'nilaiRataRata')}</td>
   139→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;color:${nilaiNum > 0 ? '#047857' : '#ccc'}">${v(r, 'skorNilaiRaport')}</td>
   140→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;color:#7c3aed">${skorLomba}</td>
   141→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;color:#7c3aed">${nilaiRataRataTKA}</td>
   142→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;color:#7c3aed">${skorPrestasiAkademik}</td>
   143→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;color:${skorNum > 0 ? '#b45309' : '#ccc'}">${v(r, 'totalNilai')}</td>
   144→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px;font-weight:${rankingTampilan === 'komposit' ? 'bold' : 'normal'};color:${skorNum > 0 ? '#b45309' : '#ccc'}">${v(r, 'skor')}</td>
   145→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;line-height:1.2;max-width:120px;overflow:hidden">${kekuranganText}</td>
   146→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;text-align:center">${v(r, 'tanggalVerif')}</td>
   147→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;text-align:center">${v(r, 'jamVerif')}</td>
   148→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;text-align:center">${v(r, 'terbitKK')}</td>
   149→      <td style="padding:4px 3px;border:1px solid #ddd;font-size:8px;text-align:center">${v(r, 'lamaKK')}</td>
   150→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:9px">${statusBadge}</td>
   151→      <td style="padding:4px 3px;border:1px solid #ddd;text-align:center;font-size:8px">${withinKuota ? '<span style="color:#047857;font-weight:bold">&#10003;</span>' : '<span style="color:#ccc">-</span>'}</td>
   152→    </tr>`
   153→  }).join('')
   154→
   155→  // Build jalur summary table
   156→  const jalurSummaryRows = Object.entries(jalurGroups).map(([jalur, s]) => {
   157→    const kuotaText = s.kuota > 0 ? String(s.kuota) : '-'
   158→    const pct = s.total > 0 ? Math.round((s.verified / s.total) * 100) : 0
   159→    return `<tr>
   160→      <td style="padding:4px 8px;border:1px solid #ddd;font-size:10px;font-weight:500">${jalur}</td>
   161→      <td style="padding:4px 8px;border:1px solid #ddd;text-align:center;font-size:10px">${kuotaText}</td>
   162→      <td style="padding:4px 8px;border:1px solid #ddd;text-align:center;font-size:10px">${s.total}</td>
   163→      <td style="padding:4px 8px;border:1px solid #ddd;text-align:center;font-size:10px;color:#047857">${s.verified}</td>
   164→      <td style="padding:4px 8px;border:1px solid #ddd;text-align:center;font-size:10px;color:#b91c1c">${s.rejected}</td>
   165→      <td style="padding:4px 8px;border:1px solid #ddd;text-align:center;font-size:10px;color:#b45309">${s.pending}</td>
   166→      <td style="padding:4px 8px;border:1px solid #ddd;text-align:center;font-size:10px">${pct}%</td>
   167→    </tr>`
   168→  }).join('')
   169→
   170→  // Total verified / rejected / pending counts
   171→  const totalVerified = reRanked.filter(r => r.verificationStatus === 'VERIFIED').length
   172→  const totalRejected = reRanked.filter(r => r.verificationStatus === 'REJECTED').length
   173→  const totalPending = reRanked.filter(r => r.verificationStatus !== 'VERIFIED' && r.verificationStatus !== 'REJECTED').length
   174→
   175→  return `<!DOCTYPE html><html><head><title>Perangkingan ${appName} - ${sortLabel}</title>
   176→    <style>
   177→      @page { size: A3 landscape; margin: 10mm; }
   178→      body { font-family: Arial, sans-serif; margin: 0; padding: 10px; font-size: 10px; }
   179→      .header { text-align: center; margin-bottom: 10px; border-bottom: 3px double #333; padding-bottom: 8px; }
   180→      .header h1 { font-size: 16px; margin: 0 0 2px 0; letter-spacing: 2px; }
   181→      .header h2 { font-size: 13px; margin: 0 0 2px 0; color: #555; }
   182→      .header h3 { font-size: 11px; margin: 0 0 2px 0; color: #777; }
   183→      .header p { font-size: 9px; color: #888; margin: 2px 0; }
   184→      .filters { display: flex; gap: 10px; justify-content: center; margin-bottom: 8px; font-size: 9px; color: #666; }
   185→      .filters span { background: #f5f5f5; padding: 2px 6px; border-radius: 4px; border: 1px solid #ddd; }
   186→      .summary-section { margin-bottom: 8px; }
   187→      .summary-section h4 { font-size: 11px; margin: 0 0 4px 0; color: #333; }
   188→      .kuota-info { text-align: center; margin-bottom: 8px; font-size: 10px; }
   189→      .kuota-info span { background: #dbeafe; color: #1e40af; padding: 2px 8px; border-radius: 4px; margin: 0 3px; }
   190→      .stats-row { display: flex; gap: 8px; justify-content: center; margin-bottom: 8px; }
   191→      .stat-box { text-align: center; padding: 4px 12px; border-radius: 6px; border: 1px solid #ddd; }
   192→      .stat-box .stat-val { font-size: 16px; font-weight: bold; }
   193→      .stat-box .stat-label { font-size: 8px; color: #666; }
   194→      table { width: 100%; border-collapse: collapse; font-size: 9px; }
   195→      th { background: #f0f4f8; padding: 4px 3px; border: 1px solid #cbd5e1; text-align: center; font-size: 8px; font-weight: 600; color: #334155; }
   196→      th.active { background: #fef3c7; color: #92400e; }
   197→      th.group-header { background: #1e293b; color: #fff; font-size: 9px; padding: 3px 5px; letter-spacing: 0.5px; }
   198→      .legend { margin-top: 8px; font-size: 9px; color: #888; text-align: center; }
   199→      .legend span { margin: 0 6px; }

   202→    </style>
   203→  </head>
   204→  <body>
   205→    <div class="header">
   206→      <h1>LAPORAN PERANGKINGAN PESERTA SPMB</h1>

   210→    </div>
   211→
   212→    <div class="filters">
   213→      <span>Jalur: ${jalurLabel}</span>
   214→      <span>Sekolah: ${sekolahLabel}</span>
   215→      <span>Jurusan: ${jurusanLabel}</span>
   216→      <span>Status: ${statusLabel}</span>
   217→    </div>
   218→
   219→    <!-- Stats Overview -->
   220→    <div class="stats-row">
   221→      <div class="stat-box" style="background:#f0f9ff;border-color:#bae6fd">
   222→        <div class="stat-val" style="color:#0369a1">${reRanked.length}</div>
   223→        <div class="stat-label">Total Pendaftar</div>
   224→      </div>
   225→      <div class="stat-box" style="background:#f0fdf4;border-color:#bbf7d0">
   226→        <div class="stat-val" style="color:#047857">${totalVerified}</div>
   227→        <div class="stat-label">Diterima</div>
   228→      </div>

   239→        <div class="stat-label">Total Kuota</div>

   241→    </div>
   242→
   243→    ${rankingKuota > 0 ? `<div class="kuota-info">${rankingKuotaPerJalur.map(kj => `<span>${kj.nama}: ${kj.kuota} (${kj.persentase}%)</span>`).join('')}</div>` : ''}
   244→

   319→      Dokumen ini dihasilkan secara otomatis oleh ${appName}${schoolName ? ' — ' + schoolName : ''} &middot; Halaman ini bersifat rahasia dan hanya untuk keperluan internal panitia SPMB
   320→    </div>
   321→  </body></html>`
   322→}

   324→/**
   325→ * Handle ranking export to Excel.
   326→ * Enhanced to match the detail level of Lembar Verifikasi.
   327→ */
   328→export function handleRankingExportExcel(params: {
   329→  rankingPreviewJalur: string
   330→  rankingTampilan: string
   331→  rankingSekolah: string
   332→  rankingJurusan: string
   333→  rankingStatus: string
   334→  rankingData: Array<Record<string, unknown>>
   335→  rankingKuota: number
   336→  rankingKuotaPerJalur: Array<{ nama: string; persentase: number; kuota: number }>
   337→  appName: string
   338→  schoolName: string
   339→}): void {
   340→  const {
   341→    rankingPreviewJalur: selectedJalur,
   342→    rankingTampilan,
   343→    rankingSekolah,
   344→    rankingJurusan,
   345→    rankingStatus,
   346→    rankingData,
   347→    rankingKuota,
   348→    rankingKuotaPerJalur,
   349→    appName,
   350→    schoolName,
   351→  } = params
   352→
   353→  const sortLabel = rankingTampilan === 'jarak' ? 'Jarak Terdekat' : rankingTampilan === 'nilai' ? 'Nilai Tertinggi' : 'Skor Komposit'
   354→  const jalurLabel = selectedJalur !== 'all' ? selectedJalur : 'Semua Jalur'
   355→
   356→  // Filter data by selected jalur
   357→  const filteredData = selectedJalur === 'all'
   358→    ? rankingData
   359→    : rankingData.filter((r: Record<string, unknown>) => (r.subJalur as string) === selectedJalur)
   360→
   361→  // Re-rank
   362→  const jalurRankCounters: Record<string, number> = {}
   363→  const reRanked = filteredData.map((r: Record<string, unknown>, idx: number) => {
   364→    const jalur = r.subJalur as string
   365→    if (!jalurRankCounters[jalur]) jalurRankCounters[jalur] = 0
   366→    jalurRankCounters[jalur]++
   367→    return { ...r, _newRank: idx + 1, _newJalurRank: jalurRankCounters[jalur] }
   368→  })
   369→
   370→  const excelData = reRanked.map((r: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) => {
   371→    const rankNum = r._newRank
   372→    const jalurRank = r._newJalurRank
   373→
   374→    const currentKuota = rankingKuotaPerJalur.find(k => {
   375→      const jalurName = (r.subJalur as string || '').toLowerCase()
   376→      return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
   377→        || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
   378→        || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   379→        || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
   380→        || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
   381→    })?.kuota || 0
   382→
   383→    const sameJalurAbove = reRanked
   384→      .filter((other: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) =>
   385→        (other.subJalur as string) === (r.subJalur as string) &&
   386→        other._newJalurRank < jalurRank
   387→      ).length
   388→
   389→    const withinKuota = currentKuota > 0 && sameJalurAbove < currentKuota
   390→
   391→    return {
   392→      'Rangking': rankNum,
   393→      'Rangking Jalur': jalurRank > 0 ? jalurRank : '-',
   394→      'Jalur': r.subJalur as string,
   395→      'No. Registrasi': v(r, 'noRegistrasi'),
   396→      'Nama': v(r, 'nama'),
   397→      'NISN': v(r, 'nisn'),
   398→      'NIK': v(r, 'nik'),
   399→      'Tanggal Lahir': v(r, 'tanggalLahir'),
   400→      'Sekolah Asal': v(r, 'namaSekolahAsal'),
   401→      'Jurusan': v(r, 'jurusan'),
   402→      'Jarak': v(r, 'lokasiJarak'),
   403→      'Skor Jarak': v(r, 'skorJarak'),
   404→      'Nilai Rata-Rata': v(r, 'nilaiRataRata'),
   405→      'Skor Nilai Raport': v(r, 'skorNilaiRaport'),
   406→      'Skor Lomba': v(r, 'skorLomba'),
   407→      'Nilai Rata-Rata TKA': v(r, 'nilaiRataRataTKA'),
   408→      'Skor Prestasi': v(r, 'skorPrestasiAkademik'),
   409→      'Skor Komposit': v(r, 'skor'),
   410→      'Kekurangan Verifikasi': v(r, 'kekuranganVerifikasi'),
   411→      'Tanggal Verif': v(r, 'tanggalVerif'),
   412→      'Jam Verif': v(r, 'jamVerif'),
   413→      'Terbit KK': v(r, 'terbitKK'),
   414→      'Lama KK': v(r, 'lamaKK'),
   415→      'Status Verifikasi': r.verificationStatus === 'VERIFIED' ? 'Diterima' : r.verificationStatus === 'REJECTED' ? 'Ditolak' : 'Menunggu',
   416→      'Masuk Kuota': withinKuota ? 'Ya' : 'Tidak',
   417→    }
   418→  })
   419→
   420→  const ws = XLSX.utils.json_to_sheet(excelData)
   421→
   422→  // Set column widths
   423→  ws['!cols'] = [
   424→    { wch: 10 }, { wch: 14 }, { wch: 20 }, { wch: 18 }, { wch: 25 },
   425→    { wch: 16 }, { wch: 18 }, { wch: 18 }, { wch: 14 }, { wch: 22 },
   426→    { wch: 14 }, { wch: 12 }, { wch: 14 }, { wch: 18 }, { wch: 18 }, { wch: 20 },
   427→    { wch: 14 }, { wch: 14 }, { wch: 28 }, { wch: 14 }, { wch: 10 }, { wch: 14 },
   428→    { wch: 12 }, { wch: 14 }, { wch: 10 },
   429→  ]
   430→
   431→  const wb = XLSX.utils.book_new()
   432→  XLSX.utils.book_append_sheet(wb, ws, 'Perangkingan')
   433→
   434→  // Build per-jalur summary for Excel
   435→  const jalurGroups: Record<string, { total: number; verified: number; rejected: number; pending: number; kuota: number }> = {}
   436→  for (const r of reRanked) {
   437→    const jalur = r.subJalur as string
   438→    if (!jalurGroups[jalur]) {
   439→      const kuota = rankingKuotaPerJalur.find(k => {
   440→        const jalurName = jalur.toLowerCase()
   441→        return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
   442→          || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
   443→          || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   444→          || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
   445→          || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
   446→      })?.kuota || 0
   447→      jalurGroups[jalur] = { total: 0, verified: 0, rejected: 0, pending: 0, kuota }
   448→    }
   449→    jalurGroups[jalur].total++
   450→    if (r.verificationStatus === 'VERIFIED') jalurGroups[jalur].verified++
   451→    else if (r.verificationStatus === 'REJECTED') jalurGroups[jalur].rejected++
   452→    else jalurGroups[jalur].pending++
   453→  }
   454→
   455→  const totalVerified = reRanked.filter(r => r.verificationStatus === 'VERIFIED').length
   456→  const totalRejected = reRanked.filter(r => r.verificationStatus === 'REJECTED').length
   457→  const totalPending = reRanked.length - totalVerified - totalRejected
   458→
   459→  // Add a summary sheet
   460→  const summaryData = [
   461→    { 'Keterangan': `LAPORAN PERANGKINGAN ${appName}${schoolName ? ' — ' + schoolName : ''}`, 'Nilai': '' },
   462→    { 'Keterangan': 'Diurutkan Berdasarkan', 'Nilai': sortLabel },
   463→    { 'Keterangan': 'Jalur', 'Nilai': jalurLabel },
   464→    { 'Keterangan': 'Sekolah Asal', 'Nilai': rankingSekolah !== 'all' ? rankingSekolah : 'Semua Sekolah' },
   465→    { 'Keterangan': 'Jurusan', 'Nilai': rankingJurusan !== 'all' ? rankingJurusan : 'Semua Jurusan' },
   466→    { 'Keterangan': 'Status Verifikasi', 'Nilai': rankingStatus !== 'all' ? (rankingStatus === 'VERIFIED' ? 'Diterima' : rankingStatus === 'REJECTED' ? 'Ditolak' : 'Menunggu') : 'Semua Status' },
   467→    { 'Keterangan': '', 'Nilai': '' },
   468→    { 'Keterangan': 'Total Pendaftar', 'Nilai': reRanked.length.toString() },
   469→    { 'Keterangan': 'Diterima', 'Nilai': totalVerified.toString() },
   470→    { 'Keterangan': 'Ditolak', 'Nilai': totalRejected.toString() },
   471→    { 'Keterangan': 'Menunggu', 'Nilai': totalPending.toString() },
   472→    { 'Keterangan': 'Total Kuota', 'Nilai': rankingKuota.toString() },
   473→    { 'Keterangan': '', 'Nilai': '' },
   474→    { 'Keterangan': 'Ringkasan Per Jalur', 'Nilai': '' },
   475→    ...Object.entries(jalurGroups).map(([jalur, s]) => ({ 'Keterangan': `  ${jalur}`, 'Nilai': `Total: ${s.total} | Diterima: ${s.verified} | Ditolak: ${s.rejected} | Menunggu: ${s.pending} | Kuota: ${s.kuota}` })),
   476→    { 'Keterangan': '', 'Nilai': '' },
   477→    { 'Keterangan': 'Kuota Per Jalur', 'Nilai': '' },
   478→    ...rankingKuotaPerJalur.map(kj => ({ 'Keterangan': `  ${kj.nama}`, 'Nilai': `${kj.kuota} (${kj.persentase}%)` })),
   479→    { 'Keterangan': '', 'Nilai': '' },
   480→    { 'Keterangan': 'Dicetak pada', 'Nilai': new Date().toLocaleDateString('id-ID', { day: '2-digit', month: 'long', year: 'numeric', hour: '2-digit', minute: '2-digit' }) },
   481→  ]
   482→  const ws2 = XLSX.utils.json_to_sheet(summaryData)
   483→  ws2['!cols'] = [{ wch: 30 }, { wch: 60 }]
   484→  XLSX.utils.book_append_sheet(wb, ws2, 'Ringkasan')
   485→
   486→  const jalurSuffix = selectedJalur !== 'all' ? `_${selectedJalur.replace(/\s+/g, '_')}` : '_Semua_Jalur'
   487→  const fileName = `Perangkingan_SPMB2026${jalurSuffix}_${sortLabel.replace(/\s+/g, '_')}_${new Date().toISOString().slice(0, 10)}.xlsx`
   488→  const wbout = XLSX.write(wb, { bookType: 'xlsx', type: 'array' })
   489→  const blob = new Blob([wbout], { type: 'application/octet-stream' })
   490→  saveAs(blob, fileName)
   491→}
