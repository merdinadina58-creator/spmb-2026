    24→function v(r: Record<string, unknown>, key: string): string {
    25→  const val = r[key]
    26→  if (val === null || val === undefined) return '-'
    27→  return String(val) || '-'
    28→}
    29→
    30→// Helper: check if a jalur name is Non-Akademik variant
    31→function isNonAkademikJalur(subJalur: string): boolean {
    32→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    33→  return lower.includes('nonakademik')
    34→}
    35→
    36→// Helper: check if a jalur name is Akademik variant (includes "Prestasi" without "Nonakademik")
    37→function isAkademikJalur(subJalur: string): boolean {
    38→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    39→  return (lower.includes('akademik') || lower === 'prestasi') && !lower.includes('nonakademik')
    40→}
    41→
    42→// Helper: check if a jalur is a prestasi type (either Akademik or Non-Akademik)
    43→function isPrestasiJalur(subJalur: string): boolean {
    44→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    45→  return lower.includes('prestasi') || lower.includes('akademik')
    46→}
    47→
    48→// Re-sort data by the appropriate prestasi score based on jalur
    49→function reSortByPrestasiJalur(data: Array<Record<string, unknown>>): Array<Record<string, unknown>> {
    50→  return [...data].sort((a, b) => {
    51→    const subJalurA = (a.subJalur as string) || ''
    52→    const subJalurB = (b.subJalur as string) || ''
    53→
    54→    const getScore = (r: Record<string, unknown>, subJalur: string): number => {
    55→      if (isNonAkademikJalur(subJalur)) {
    56→        return (r._skorPrestasiNonAkademikNum as number) || -1
    57→      }
    58→      if (isAkademikJalur(subJalur)) {
    59→        return (r._skorPrestasiAkademikNum as number) || -1
    60→      }
    61→      // Fallback
    62→      const akdNum = (r._skorPrestasiAkademikNum as number) || -1
    63→      const nonAkdNum = (r._skorPrestasiNonAkademikNum as number) || -1
    64→      return akdNum > 0 ? akdNum : nonAkdNum
    65→    }
    66→
    67→    const scoreA = getScore(a, subJalurA)
    68→    const scoreB = getScore(b, subJalurB)
    69→
    70→    if (scoreA > 0 && scoreB > 0) return scoreB - scoreA
    71→    if (scoreA > 0) return -1
    72→    if (scoreB > 0) return 1
    73→    return 0
    74→  })
    75→}
