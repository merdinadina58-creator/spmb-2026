     4→/**
     5→ * Safely get a string value from ranking data record
     6→ */
     7→function v(r: Record<string, unknown>, key: string): string {
     8→  const val = r[key]
     9→  if (val === null || val === undefined) return '-'
    10→  return String(val) || '-'
    11→}

    13→// Helper: check if a jalur name is Non-Akademik variant
    14→function isNonAkademikJalur(subJalur: string): boolean {
    15→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    16→  return lower.includes('nonakademik')
    17→}
    18→
    19→// Helper: check if a jalur name is Akademik variant (includes "Prestasi" without "Nonakademik")
    20→function isAkademikJalur(subJalur: string): boolean {
    21→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    22→  return (lower.includes('akademik') || lower === 'prestasi') && !lower.includes('nonakademik')
    23→}
    24→
    25→// Helper: check if a jalur is a prestasi type (either Akademik or Non-Akademik)
    26→function isPrestasiJalur(subJalur: string): boolean {
    27→  const lower = subJalur.toLowerCase().replace(/[^a-z0-9]/g, '')
    28→  return lower.includes('prestasi') || lower.includes('akademik')
    29→}
    30→
    31→// Re-sort data by the appropriate prestasi score based on jalur
    32→// Prestasi Akademik → sort by _skorPrestasiAkademikNum (desc)
    33→// Prestasi Non Akademik → sort by _skorPrestasiNonAkademikNum (desc)
    34→function reSortByPrestasiJalur(data: Array<Record<string, unknown>>): Array<Record<string, unknown>> {
    35→  return [...data].sort((a, b) => {
    36→    const subJalurA = (a.subJalur as string) || ''
    37→    const subJalurB = (b.subJalur as string) || ''
    38→
    39→    // Get the appropriate score for each record based on its jalur
    40→    const getScore = (r: Record<string, unknown>, subJalur: string): number => {
    41→      if (isNonAkademikJalur(subJalur)) {
    42→        return (r._skorPrestasiNonAkademikNum as number) || -1
    43→      }
    44→      if (isAkademikJalur(subJalur)) {
    45→        return (r._skorPrestasiAkademikNum as number) || -1
    46→      }
    47→      // Fallback for other jalur: use the jalur-aware prestasi score
    48→      return getPrestasiNumValue(r)
    49→    }
    50→
    51→    const scoreA = getScore(a, subJalurA)
    52→    const scoreB = getScore(b, subJalurB)
    53→
    54→    // Both have data: sort descending (highest first)
    55→    if (scoreA > 0 && scoreB > 0) return scoreB - scoreA
    56→    // Only one has data: data comes first
    57→    if (scoreA > 0) return -1
    58→    if (scoreB > 0) return 1
    59→    return 0
    60→  })
    61→}
    62→
    63→// Helper: get prestasi score for display from a record
    64→function getPrestasiDisplayValue(r: Record<string, unknown>): string {
    65→  const subJalur = (r.subJalur as string) || ''
    66→  if (isNonAkademikJalur(subJalur)) {
    67→    return v(r, 'skorPrestasiNonAkademik') !== '-' ? v(r, 'skorPrestasiNonAkademik') : v(r, 'skorPrestasiAkademik')
    68→  }
    69→  return v(r, 'skorPrestasiAkademik') !== '-' ? v(r, 'skorPrestasiAkademik') : v(r, 'skorPrestasiNonAkademik')
    70→}
    71→
    72→// Helper: get numeric prestasi score from a record
    73→function getPrestasiNumValue(r: Record<string, unknown>): number {
    74→  const subJalur = (r.subJalur as string) || ''
    75→  const akdNum = (r._skorPrestasiAkademikNum as number) || -1
    76→  const nonAkdNum = (r._skorPrestasiNonAkademikNum as number) || -1
    77→  if (isNonAkademikJalur(subJalur)) {
    78→    return nonAkdNum > 0 ? nonAkdNum : akdNum
    79→  }
    80→  return akdNum > 0 ? akdNum : nonAkdNum
    81→}

    83→/**
    84→ * Generate print HTML content for ranking report.
    85→ * Designed to match the detail level of Lembar Verifikasi.
    86→ */
    87→export function getRankingPrintHTML(params: {

   103→    rankingJurusan,
   104→    rankingStatus,
   105→    rankingData,
   106→    rankingKuota,

   108→    appName,
   109→    schoolName,
   110→  } = params
   111→
   112→  const sortLabel = rankingTampilan === 'jarak' ? 'Jarak Terdekat' : rankingTampilan === 'nilai' ? 'Nilai Tertinggi' : rankingTampilan === 'komposit' ? 'Skor Komposit' : 'Skor Prestasi'
   113→  const prestasiNote = isPrestasiJalur(selectedJalur) ? ' (diurutkan berdasarkan Skor Prestasi sesuai jalur)' : ''
   114→  const jalurLabel = selectedJalur !== 'all' ? selectedJalur : 'Semua Jalur'
   115→  const sekolahLabel = rankingSekolah !== 'all' ? rankingSekolah : 'Semua Sekolah'

   123→
   124→  // Re-sort by appropriate prestasi score based on jalur for printing
   125→  // This ensures the print always ranks by the correct prestasi score
   126→  // regardless of the current tampilan mode

   141→  function findKuota(subJalur: string): number {

   146→        || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   147→        || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))

   151→
   152→  // Build per-jalur summary stats

   162→    else jalurGroups[jalur].pending++

   180→        (other.subJalur as string) === (r.subJalur as string) &&
   181→        other._newJalurRank < jalurRank

   187→    const rowBg = withinKuota && !isVerified ? '#f0fdf4' : isVerified ? '#f0fdf4' : r.verificationStatus === 'REJECTED' ? '#fef2f2' : ''
   188→    const rankBg = rankNum === 1 ? '#fbbf24' : rankNum === 2 ? '#d1d5db' : rankNum === 3 ? '#b45309' : withinKuota ? '#d1fae5' : '#f3f4f6'

   196→        : '<span style="background:#fef3c7;color:#b45309;padding:1px 6px;border-radius:8px;font-size:9px">Menunggu</span>'

   406→/**
   407→ * Handle ranking export to Excel.
   408→ * Enhanced to match the detail level of Lembar Verifikasi.
   409→ */
   410→export function handleRankingExportExcel(params: {
   411→  rankingPreviewJalur: string
   412→  rankingTampilan: string
   413→  rankingSekolah: string
   414→  rankingJurusan: string
   415→  rankingStatus: string
   416→  rankingData: Array<Record<string, unknown>>
   417→  rankingKuota: number
   418→  rankingKuotaPerJalur: Array<{ nama: string; persentase: number; kuota: number }>
   419→  appName: string
   420→  schoolName: string
   421→}): void {
   422→  const {
   423→    rankingPreviewJalur: selectedJalur,
   424→    rankingTampilan,
   425→    rankingSekolah,
   426→    rankingJurusan,
   427→    rankingStatus,
   428→    rankingData,
   429→    rankingKuota,
   430→    rankingKuotaPerJalur,
   431→    appName,
   432→    schoolName,
   433→  } = params
   434→
   435→  const sortLabel = rankingTampilan === 'jarak' ? 'Jarak Terdekat' : rankingTampilan === 'nilai' ? 'Nilai Tertinggi' : rankingTampilan === 'komposit' ? 'Skor Komposit' : 'Skor Prestasi'
   436→  const prestasiNote = isPrestasiJalur(selectedJalur) ? ' (diurutkan berdasarkan Skor Prestasi sesuai jalur)' : ''
   437→  const jalurLabel = selectedJalur !== 'all' ? selectedJalur : 'Semua Jalur'
   438→
   439→  // Filter data by selected jalur
   440→  const filteredData = selectedJalur === 'all'
   441→    ? rankingData
   442→    : rankingData.filter((r: Record<string, unknown>) => (r.subJalur as string) === selectedJalur)
   443→
   444→  // Re-sort by appropriate prestasi score based on jalur for export
   445→  // This ensures the Excel export always ranks by the correct prestasi score
   446→  // regardless of the current tampilan mode
   447→  const sortedForExport = isPrestasiJalur(selectedJalur)
   448→    ? reSortByPrestasiJalur(filteredData)
   449→    : filteredData
   450→
   451→  // Re-rank
   452→  const jalurRankCounters: Record<string, number> = {}
   453→  const reRanked = sortedForExport.map((r: Record<string, unknown>, idx: number) => {
   454→    const jalur = r.subJalur as string
   455→    if (!jalurRankCounters[jalur]) jalurRankCounters[jalur] = 0
   456→    jalurRankCounters[jalur]++
   457→    return { ...r, _newRank: idx + 1, _newJalurRank: jalurRankCounters[jalur] }
   458→  })
   459→
   460→  const excelData = reRanked.map((r: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) => {
   461→    const rankNum = r._newRank
   462→    const jalurRank = r._newJalurRank
   463→
   464→    const currentKuota = rankingKuotaPerJalur.find(k => {
   465→      const jalurName = (r.subJalur as string || '').toLowerCase()
   466→      return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
   467→        || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
   468→        || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   469→        || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
   470→        || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
   471→    })?.kuota || 0
   472→
   473→    const sameJalurAbove = reRanked
   474→      .filter((other: Record<string, unknown> & { _newRank: number; _newJalurRank: number }) =>
   475→        (other.subJalur as string) === (r.subJalur as string) &&
   476→        other._newJalurRank < jalurRank
   477→      ).length
   478→
   479→    const withinKuota = currentKuota > 0 && sameJalurAbove < currentKuota
   480→
   481→    return {
   482→      'Rangking': rankNum,
   483→      'Rangking Jalur': jalurRank > 0 ? jalurRank : '-',
   484→      'Jalur': r.subJalur as string,
   485→      'No. Registrasi': v(r, 'noRegistrasi'),
   486→      'Nama': v(r, 'nama'),
   487→      'NISN': v(r, 'nisn'),
   488→      'NIK': v(r, 'nik'),
   489→      'Tanggal Lahir': v(r, 'tanggalLahir'),
   490→      'Sekolah Asal': v(r, 'namaSekolahAsal'),
   491→      'Jurusan': v(r, 'jurusan'),
   492→      'Jarak': v(r, 'lokasiJarak'),
   493→      'Skor Jarak': v(r, 'skorJarak'),
   494→      'Nilai Rata-Rata': v(r, 'nilaiRataRata'),
   495→      'Skor Nilai Raport': v(r, 'skorNilaiRaport'),
   496→      'Skor Lomba': v(r, 'skorLomba'),
   497→      'Nilai Rata-Rata TKA': v(r, 'nilaiRataRataTKA'),
   498→      'Skor Prestasi': v(r, 'skorPrestasiAkademik'),
   499→      'Skor Prestasi Akademik': (r.skorPrestasiAkademik as string) || '-',
   500→      'Skor Prestasi Non Akademik': (r.skorPrestasiNonAkademik as string) || '-',
   501→      'Skor Komposit': v(r, 'skor'),
   502→      'Kekurangan Verifikasi': v(r, 'kekuranganVerifikasi'),
   503→      'Tanggal Verif': v(r, 'tanggalVerif'),
   504→      'Jam Verif': v(r, 'jamVerif'),
   505→      'Terbit KK': v(r, 'terbitKK'),
   506→      'Lama KK': v(r, 'lamaKK'),
   507→      'Status Verifikasi': r.verificationStatus === 'VERIFIED' ? 'Diterima' : r.verificationStatus === 'REJECTED' ? 'Ditolak' : 'Menunggu',
   508→      'Masuk Kuota': withinKuota ? 'Ya' : 'Tidak',
   509→    }
   510→  })
   511→
   512→  const ws = XLSX.utils.json_to_sheet(excelData)
   513→
   514→  // Set column widths
   515→  ws['!cols'] = [
   516→    { wch: 10 }, { wch: 14 }, { wch: 20 }, { wch: 18 }, { wch: 25 },
   517→    { wch: 16 }, { wch: 18 }, { wch: 18 }, { wch: 14 }, { wch: 22 },
   518→    { wch: 14 }, { wch: 12 }, { wch: 14 }, { wch: 18 }, { wch: 18 }, { wch: 20 },
   519→    { wch: 14 }, { wch: 14 }, { wch: 28 }, { wch: 14 }, { wch: 10 }, { wch: 14 },
   520→    { wch: 12 }, { wch: 14 }, { wch: 10 },
   521→  ]
   522→
   523→  const wb = XLSX.utils.book_new()
   524→  XLSX.utils.book_append_sheet(wb, ws, 'Perangkingan')
   525→
   526→  // Build per-jalur summary for Excel
   527→  const jalurGroups: Record<string, { total: number; verified: number; rejected: number; pending: number; kuota: number }> = {}
   528→  for (const r of reRanked) {
   529→    const jalur = r.subJalur as string
   530→    if (!jalurGroups[jalur]) {
   531→      const kuota = rankingKuotaPerJalur.find(k => {
   532→        const jalurName = jalur.toLowerCase()
   533→        return k.nama.toLowerCase().includes(jalurName) || jalurName.includes(k.nama.toLowerCase())
   534→          || (k.nama.toLowerCase().includes('prestasi') && jalurName.includes('prestasi'))
   535→          || (k.nama.toLowerCase().includes('domisili') && jalurName.includes('domisili'))
   536→          || (k.nama.toLowerCase().includes('mutasi') && jalurName.includes('mutasi'))
   537→          || (k.nama.toLowerCase().includes('afirmasi') && (jalurName.includes('keluarga') || jalurName.includes('ktm')))
   538→      })?.kuota || 0
   539→      jalurGroups[jalur] = { total: 0, verified: 0, rejected: 0, pending: 0, kuota }
   540→    }
   541→    jalurGroups[jalur].total++
   542→    if (r.verificationStatus === 'VERIFIED') jalurGroups[jalur].verified++
   543→    else if (r.verificationStatus === 'REJECTED') jalurGroups[jalur].rejected++
   544→    else jalurGroups[jalur].pending++
   545→  }
   546→
   547→  const totalVerified = reRanked.filter(r => r.verificationStatus === 'VERIFIED').length
   548→  const totalRejected = reRanked.filter(r => r.verificationStatus === 'REJECTED').length
   549→  const totalPending = reRanked.length - totalVerified - totalRejected
   550→
   551→  // Add a summary sheet
   552→  const summaryData = [
   553→    { 'Keterangan': `LAPORAN PERANGKINGAN ${appName}${schoolName ? ' — ' + schoolName : ''}`, 'Nilai': '' },
   554→    { 'Keterangan': 'Diurutkan Berdasarkan', 'Nilai': `${sortLabel}${prestasiNote}` },
   555→    { 'Keterangan': 'Jalur', 'Nilai': jalurLabel },
   556→    { 'Keterangan': 'Sekolah Asal', 'Nilai': rankingSekolah !== 'all' ? rankingSekolah : 'Semua Sekolah' },
   557→    { 'Keterangan': 'Jurusan', 'Nilai': rankingJurusan !== 'all' ? rankingJurusan : 'Semua Jurusan' },
   558→    { 'Keterangan': 'Status Verifikasi', 'Nilai': rankingStatus !== 'all' ? (rankingStatus === 'VERIFIED' ? 'Diterima' : rankingStatus === 'REJECTED' ? 'Ditolak' : 'Menunggu') : 'Semua Status' },
   559→    { 'Keterangan': '', 'Nilai': '' },
   560→    { 'Keterangan': 'Total Pendaftar', 'Nilai': reRanked.length.toString() },
   561→    { 'Keterangan': 'Diterima', 'Nilai': totalVerified.toString() },
   562→    { 'Keterangan': 'Ditolak', 'Nilai': totalRejected.toString() },
   563→    { 'Keterangan': 'Menunggu', 'Nilai': totalPending.toString() },
   564→    { 'Keterangan': 'Total Kuota', 'Nilai': rankingKuota.toString() },
   565→    { 'Keterangan': '', 'Nilai': '' },
   566→    { 'Keterangan': 'Ringkasan Per Jalur', 'Nilai': '' },
   567→    ...Object.entries(jalurGroups).map(([jalur, s]) => ({ 'Keterangan': `  ${jalur}`, 'Nilai': `Total: ${s.total} | Diterima: ${s.verified} | Ditolak: ${s.rejected} | Menunggu: ${s.pending} | Kuota: ${s.kuota}` })),
   568→    { 'Keterangan': '', 'Nilai': '' },
   569→    { 'Keterangan': 'Kuota Per Jalur', 'Nilai': '' },
   570→    ...rankingKuotaPerJalur.map(kj => ({ 'Keterangan': `  ${kj.nama}`, 'Nilai': `${kj.kuota} (${kj.persentase}%)` })),
   571→    { 'Keterangan': '', 'Nilai': '' },
   572→    { 'Keterangan': 'Dicetak pada', 'Nilai': new Date().toLocaleDateString('id-ID', { day: '2-digit', month: 'long', year: 'numeric', hour: '2-digit', minute: '2-digit' }) },
   573→  ]
   574→  const ws2 = XLSX.utils.json_to_sheet(summaryData)
   575→  ws2['!cols'] = [{ wch: 30 }, { wch: 60 }]
   576→  XLSX.utils.book_append_sheet(wb, ws2, 'Ringkasan')
   577→
   578→  const jalurSuffix = selectedJalur !== 'all' ? `_${selectedJalur.replace(/\s+/g, '_')}` : '_Semua_Jalur'
   579→  const fileName = `Perangkingan_SPMB2026${jalurSuffix}_${sortLabel.replace(/\s+/g, '_')}_${new Date().toISOString().slice(0, 10)}.xlsx`
   580→  const wbout = XLSX.write(wb, { bookType: 'xlsx', type: 'array' })
   581→  const blob = new Blob([wbout], { type: 'application/octet-stream' })
   582→  saveAs(blob, fileName)
   583→}
