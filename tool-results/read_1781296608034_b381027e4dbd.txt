  1249→  const parseCSVClientSide = (text: string) => {
  1250→    const lines = text.trim().split('\n')
  1251→    if (lines.length < 2) return []
  1252→    const headers = lines[0].split(',').map(h => h.replace(/^"|"$/g, '').trim())
  1253→    const rows: Record<string, string>[] = []
  1254→    let skippedRows = 0
  1255→    for (let i = 1; i < lines.length; i++) {
  1256→      if (!lines[i].trim()) continue // skip empty lines
  1257→      const values: string[] = []
  1258→      let current = ''
  1259→      let inQuotes = false
  1260→      for (const char of lines[i]) {
  1261→        if (char === '"') { inQuotes = !inQuotes } else if (char === ',' && !inQuotes) { values.push(current.trim()); current = '' } else { current += char }
  1262→      }
  1263→      values.push(current.trim())
  1264→      // Accept rows with close column counts (allow slight mismatches by padding)
  1265→      if (values.length >= headers.length - 2 && values.length <= headers.length + 2) {
  1266→        const row: Record<string, string> = {}
  1267→        headers.forEach((header, index) => { row[header] = values[index] || '' })
  1268→        rows.push(row)
  1269→      } else {
  1270→        skippedRows++
  1271→      }
  1272→    }
  1273→    if (skippedRows > 0) {
  1274→      toast({ title: 'Perhatian', description: `${skippedRows} baris CSV dilewati karena format tidak valid`, variant: 'destructive' })
  1275→    }
  1276→    return rows
  1277→  }
